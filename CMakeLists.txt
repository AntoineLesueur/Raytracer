cmake_minimum_required(VERSION 4.3)
project(raytracer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OpenCV_DIR "/opt/opencv-ffmpeg/lib64/cmake/opencv4")
find_package(OpenCV REQUIRED)
find_package(SFML 2 REQUIRED COMPONENTS graphics window system)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.cpp")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(raytracer ${SOURCES})
target_include_directories(raytracer PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(raytracer PRIVATE ${OpenCV_LIBS} sfml-graphics sfml-window sfml-system)
